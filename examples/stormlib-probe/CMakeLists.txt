cmake_minimum_required(VERSION 3.15)
project(stormlib_probe LANGUAGES CXX)
if(NOT STORM_SOURCE OR NOT STORM_LIBRARY)
  message(FATAL_ERROR "Pass STORM_SOURCE and STORM_LIBRARY from the pinned v9.40 build")
endif()
add_executable(stormlib-probe main.cpp)
target_compile_features(stormlib-probe PRIVATE cxx_std_17)
target_compile_definitions(stormlib-probe PRIVATE UNICODE _UNICODE __STORMLIB_NO_STATIC_LINK__)
target_compile_options(stormlib-probe PRIVATE /utf-8)
target_include_directories(stormlib-probe PRIVATE "${STORM_SOURCE}/src")
target_link_libraries(stormlib-probe PRIVATE "${STORM_LIBRARY}")
